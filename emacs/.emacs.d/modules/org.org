      '("xelatex -shell-escape -interaction nonstopmode -output-directory %o %f"
        "xelatex -shell-escape -interaction nonstopmode -output-directory %o %f"))

      '(:foreground default :background default :scale 1.5

  :ensure t
  :custom
  (org-roam-directory (file-truename "~/org"))
  (org-roam-db-location "~/.emacs.d/org-roam.db")
  (org-roam-capture-templates
   '(("d" "default" plain "%?"
      :if-new (file+head "%<%Y%m%d%H%M%S>-${slug}.org"
                 "#+title: ${title}\n#+author: singdile\n#+date: %<%Y-%m-%d>\n\n")
      :unnarrowed t)))
  :config
  (org-roam-db-autosync-mode 1)
  :bind (("C-c n f" . org-roam-node-find)
         ("C-c n i" . org-roam-node-insert)
         ("C-c n l" . org-roam-buffer-toggle)))

** org-format-inline-code：修复 =code= 前后空格

#+begin_src emacs-lisp
(defun my/org-fix-inline-code-spaces ()
  (interactive)
  (let ((start (if (use-region-p) (region-beginning) (save-excursion (backward-paragraph) (point))))
        (end (if (use-region-p) (region-end) (save-excursion (forward-paragraph) (point)))))
    (save-excursion
      (goto-char start)
      (while (re-search-forward "=\\([^= \n\t]+\\)=" end t)
        (let ((match-start (match-beginning 0))
              (match-end (match-end 0))
              (code-content (match-string 1)))
          (unless (or (= match-start (point-min))
                      (memq (char-before match-start) '(?  ?\t ?\n)))
            (goto-char match-start)
            (insert " ")
            (setq match-end (1+ match-end)))
          (unless (or (= match-end (point-max))
                      (memq (char-after match-end) '(?  ?\t ?\n)))
            (goto-char match-end)
            (insert " "))))))
  (message "已修复 =code= 前后空格"))
#+end_src
